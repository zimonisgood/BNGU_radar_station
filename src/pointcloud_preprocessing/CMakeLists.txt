cmake_minimum_required(VERSION 3.16)
project(pointcloud_preprocessing)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2 REQUIRED)
add_library(field_filter INTERFACE)
target_include_directories(field_filter INTERFACE include)
target_compile_features(field_filter INTERFACE cxx_std_17)
ament_target_dependencies(field_filter INTERFACE sensor_msgs tf2)
add_executable(field_filter_node src/field_filter_node.cpp)
target_link_libraries(field_filter_node field_filter)
ament_target_dependencies(field_filter_node rclcpp)
target_compile_options(field_filter_node PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS field_filter_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_field_filter test/test_field_filter.cpp)
  target_link_libraries(test_field_filter field_filter)
endif()
ament_package()
