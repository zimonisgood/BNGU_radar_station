cmake_minimum_required(VERSION 3.16)
project(pfa_radar_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
list(APPEND CMAKE_MODULE_PATH ${CMAKE_CURRENT_SOURCE_DIR}/cmake)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs highgui videoio dnn)
find_package(yaml-cpp REQUIRED)
option(RADAR_WITH_TENSORRT "Enable TensorRT" ON)
option(RADAR_WITH_MVS "Enable Hikvision MVS" ON)
add_library(radar_core
  src/config/config.cpp
  src/infer/preprocess.cpp
  src/infer/detector_factory.cpp
  src/camera/frame_source.cpp
  src/geo/calibration.cpp
  src/pipeline/roi_packer.cpp
)
target_include_directories(radar_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(radar_core PUBLIC ${OpenCV_LIBS} yaml-cpp)
if(RADAR_WITH_TENSORRT)
  find_package(CUDAToolkit)
  find_package(TensorRT)
  if(TensorRT_FOUND AND CUDAToolkit_FOUND)
    target_compile_definitions(radar_core PUBLIC RADAR_WITH_TENSORRT)
    target_include_directories(radar_core PUBLIC ${TENSORRT_INCLUDE_DIRS} ${CUDAToolkit_INCLUDE_DIRS})
    target_link_libraries(radar_core PUBLIC ${TENSORRT_LIBRARIES} CUDA::cudart)
    message(STATUS "TensorRT ON")
  else()
    message(WARNING "TensorRT/CUDA missing; OpenCV DNN only")
  endif()
endif()
if(RADAR_WITH_MVS)
  find_package(MVS)
  if(MVS_FOUND)
    target_sources(radar_core PRIVATE src/camera/hik_camera.cpp)
    target_compile_definitions(radar_core PUBLIC RADAR_WITH_MVS)
    target_include_directories(radar_core PUBLIC ${MVS_INCLUDE_DIRS})
    target_link_libraries(radar_core PUBLIC ${MVS_LIBRARIES})
    message(STATUS "MVS ON")
  else()
    message(WARNING "MVS missing")
  endif()
endif()
add_executable(radar_main apps/radar_main.cpp)
target_link_libraries(radar_main PRIVATE radar_core)
add_executable(hik_viewer apps/hik_viewer.cpp)
target_link_libraries(hik_viewer PRIVATE radar_core)
add_executable(yolo_bench apps/yolo_bench.cpp)
target_link_libraries(yolo_bench PRIVATE radar_core)
